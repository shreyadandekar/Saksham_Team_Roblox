<script>
    (function () {
      let baseTier = 35;
      let progMultiplier = 1.0;
      let isLeased = false;

      const totalDisplay = document.getElementById('total-deposit-display');
      const wordsDisplay = document.getElementById('total-words-display');
      const baseNormText = document.getElementById('base-norm-text');
      const factorNormText = document.getElementById('factor-norm-text');
      const leaseBox = document.getElementById('toggle-lease-box');

      function calculate() {
        const factor = isLeased ? 2.0 : 1.0;
        const totalLakhs = baseTier * progMultiplier * factor;
        const amountINR = totalLakhs * 100000;

        totalDisplay.textContent = '₹' + amountINR.toLocaleString('en-IN');
        baseNormText.textContent = '₹' + (baseTier * progMultiplier).toFixed(2) + ' Lakhs';
        
        if (isLeased) {
          factorNormText.textContent = '2.0x (Private Leased)';
          factorNormText.className = 'text-amber-400 font-semibold';
        } else {
          factorNormText.textContent = '1.0x (Trust Owned)';
          factorNormText.className = 'text-emerald-400 font-semibold';
        }

        if (totalLakhs === 15) {
          wordsDisplay.textContent = 'Rupees Fifteen Lakhs Only';
        } else if (totalLakhs === 30) {
          wordsDisplay.textContent = 'Rupees Thirty Lakhs Only';
        } else if (totalLakhs === 35) {
          wordsDisplay.textContent = 'Rupees Thirty-Five Lakhs Only';
        } else if (totalLakhs === 70) {
          wordsDisplay.textContent = 'Rupees Seventy Lakhs Only';
        } else {
          wordsDisplay.textContent = 'Rupees ' + totalLakhs.toFixed(2) + ' Lakhs Only';
        }
      }

      // Program buttons
      const progButtons = document.querySelectorAll('.calc-prog');
      progButtons.forEach(btn => {
        btn.addEventListener('click', () => {
          progButtons.forEach(b => {
            b.classList.remove('bg-[#0B2545]', 'text-white', 'shadow-xs');
            b.classList.add('bg-slate-100', 'text-slate-700');
          });
          btn.classList.add('bg-[#0B2545]', 'text-white', 'shadow-xs');
          btn.classList.remove('bg-slate-100', 'text-slate-700');
          progMultiplier = parseFloat(btn.getAttribute('data-mult') || 1.0);
          calculate();
        });
      });

      // Tier Radio Buttons
      const tierRadios = document.querySelectorAll('input[name="academic-tier"]');
      tierRadios.forEach(radio => {
        radio.addEventListener('change', (e) => {
          baseTier = parseFloat(e.target.value);
          // Highlight label styling
          tierRadios.forEach(r => {
            const p = r.closest('label');
            if (p) {
              p.classList.remove('border-2', 'border-[#0B2545]', 'bg-blue-50/40');
              p.classList.add('border-slate-200', 'bg-slate-50/50');
            }
          });
          const parent = e.target.closest('label');
          if (parent) {
            parent.classList.remove('border-slate-200', 'bg-slate-50/50');
            parent.classList.add('border-2', 'border-[#0B2545]', 'bg-blue-50/40');
          }
          calculate();
        });
      });

      // Lease Toggle
      if (leaseBox) {
        leaseBox.addEventListener('change', (e) => {
          isLeased = e.target.checked;
          calculate();
        });
      }

      // Verify button simulation
      const verifyBtn = document.getElementById('fdr-verify-btn');
      const resultCard = document.getElementById('verified-result-card');
      if (verifyBtn && resultCard) {
        verifyBtn.addEventListener('click', () => {
          verifyBtn.innerHTML = '<span class="material-symbols-outlined text-[18px] animate-spin">refresh</span><span>Verifying...</span>';
          setTimeout(() => {
            verifyBtn.innerHTML = '<span class="material-symbols-outlined text-[18px]">gavel</span><span>Verify Instrument</span>';
            resultCard.classList.add('ring-2', 'ring-[#A63B00]');
            setTimeout(() => resultCard.classList.remove('ring-2', 'ring-[#A63B00]'), 1000);
          }, 400);
        });
      }
    })();
  </script>
